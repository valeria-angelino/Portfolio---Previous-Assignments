////////////////////// Beta Convergence


gen log_gdp_pc = log(regional_gdp_cap_1990)

keep if year >= 1950

xtset _ID year

bysort _ID (year): gen growth_regional_gdp_cap_1990 = (regional_gdp_cap_1990[_n]-regional_gdp_cap_1990[_n-1]) / (regional_gdp_cap_1990[_n-1] * (year - year[_n-1]))

bysort _ID: egen mean_growth_rate = mean(growth_regional_gdp_cap_1990)

keep if year == 1950

		///// Plot with labels - BETA convergence

gen region_labels = region if region == "Southern Denmark" | region == "Sjælland" | region == "Hovedstaden" | region == "Nordjylland"| region == "Midtjylland"

twoway (scatter mean_growth_rate log_gdp_pc, mlabel(region_labels) mcolor(purple) mlabcolor(purple)) ///
 (lfit mean_growth_rate log_gdp_pc, lcolor(lavender)), ///
title("Beta-Convergence", size(large)) ///
subtitle("Regions in Denmark 1950-2015", size(medium)) ///
xlabel(8.6(0.1)9.2, labsize(small)) ///
ylabel(.018(.001).024, labsize(small)) ///
 ytitle("Average growth rate since 1950", size(small)) ///
xtitle("Initial GDP per capita (log)", size(small)) ///
 legend(off) ///
	note(Source: Rosés-Wolf (2020), size(vsmall) position(7))

graph export Beta_Convergence.png, replace width(2000)


////////////////////// Sigma Convergence by Jonathan

keep if year >= 1950
xtset _ID year

///Population-weighted sigma convergence
bysort year: egen total_pop_year = sum(regional_population)
gen pop_weight = regional_population / total_pop_year

bysort year: egen w_mean_gdp = sum(regional_gdp_cap_1990 * pop_weight)

gen sq_dev = (regional_gdp_cap_1990 - w_mean_gdp)^2
bysort year: egen w_var_gdp = sum(sq_dev * pop_weight)
gen w_sd_gdp = sqrt(w_var_gdp)

///Population-weighted coefficient of variation
gen w_cv_gdp = w_sd_gdp / w_mean_gdp

///Weighted sigma convergence plot
twoway (connected w_cv_gdp year, lcolor(blue) mcolor(blue) msymbol(circle)) ///
       (lfit w_cv_gdp year, lcolor(red)), ///
       title("Population-weighted Sigma-convergence") subtitle("Denmark 1950 - 2015") ///
       ytitle("Population-weighted CV of GDP per capita") xtitle("Year") ///
       legend(label(1 "Annual weighted CV") label(2 "Linear trend"))

/// Compare weighted and unweighted measures


bysort year: egen mean_gdp = mean(regional_gdp_cap_1990)
bysort year: egen sd_gdp = sd(regional_gdp_cap_1990)
gen cv_gdp = sd_gdp / mean_gdp

//// Final Plot

twoway (connected w_cv_gdp year, lcolor(orange) mcolor(orange) msymbol(circle)) ///
       (connected cv_gdp year, lcolor(purple) mcolor(purple) msymbol(diamond)) ///
      title("Sigma-convergence: Weighted vs Unweighted") ///
	  subtitle("Denmark 1900 - 2015") ///
       ytitle("Coefficient of variation"), size(small) ///
	   xtitle("Year"), size(small) ///
       legend(label(1 "Population-weighted CV") label(2 "Unweighted CV")) ///
	   note(Source: Rosés-Wolf (2020), size(vsmall) position(7))

twoway (connected w_cv_gdp year, lcolor(orange) mcolor(orange) msymbol(circle) msize(vsmall)) ///
       (connected cv_gdp year, lcolor(purple) mcolor(purple) msymbol(diamond) msize(vsmall)), ///
       title("Sigma-convergence: Weighted vs Unweighted") ///
       subtitle("Denmark 1900 - 2015") ///
       ytitle("Coefficient of Variation") ///
       xtitle("Year") ///
       legend(order(1 "Population-weighted CV" 2 "Unweighted CV") ///
              position(5) ring(0) size(vsmall)) ///
       note("Source: Rosés-Wolf (2020)", size(vsmall) position(7))
	   
graph export Sigma_Convergence_Real.png, replace width(2000)
